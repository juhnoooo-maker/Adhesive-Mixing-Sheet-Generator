import numpy as np import json # Define training dataset based on test1.xlsx data (T1: A:10, B:5, C:1 per 500M @ 1000mm) # Inputs: [meters / 2000.0, width / 1000.0] # Outputs for T1: [A, B, C] # Outputs for T2: [B, E, G] X_train = [] Y_t1 = [] Y_t2 = [] for m in range(100, 2100, 100): for w in [900, 1000, 1100]: area_factor = (w / 1000.0) * (m / 500.0) X_train.append([m / 2000.0, w / 1000.0]) Y_t1.append([10.0 * area_factor, 5.0 * area_factor, 1.0 * area_factor]) Y_t2.append([10.0 * area_factor, 5.0 * area_factor, 1.0 * area_factor]) X = np.array(X_train) Y1 = np.array(Y_t1) # Let's train a simple 2-layer MLP (Input 2 -> Hidden 8 -> Output 3) using numpy gradient descent np.random.seed(42) # T1 weights W1 = np.random.randn(2, 8) * 0.1 b1 = np.zeros((1, 8)) W2 = np.random.randn(8, 3) * 0.1 b2 = np.zeros((1, 3)) # Training loop lr = 0.1 for epoch in range(5000): # Forward z1 = np.dot(X, W1) + b1 a1 = np.maximum(0, z1) # ReLU z2 = np.dot(a1, W2) + b2 loss = np.mean((z2 - Y1) ** 2) # Backward dz2 = 2 * (z2 - Y1) / len(X) dW2 = np.dot(a1.T, dz2) db2 = np.sum(dz2, axis=0, keepdims=True) da1 = np.dot(dz2, W2.T) dz1 = da1 * (z1 > 0) dW1 = np.dot(X.T, dz1) db1 = np.sum(dz1, axis=0, keepdims=True) W1 -= lr * dW1 b1 -= lr * db1 W2 -= lr * dW2 b2 -= lr * db2 print(f"Final Loss: {loss:.6f}") # Test prediction for 750M, 1000mm x_test = np.array([[750 / 2000.0, 1000 / 1000.0]]) h = np.maximum(0, np.dot(x_test, W1) + b1) pred = np.dot(h, W2) + b2 print("Prediction for 750M 1000mm:", pred[0]) # Export model parameters to JSON format for JS model_json = { "W1": W1.tolist(), "b1": b1.tolist()[0], "W2": W2.tolist(), "b2": b2.tolist()[0] } print(json.dumps(model_json))
